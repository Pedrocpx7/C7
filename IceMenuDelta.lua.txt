local function arrastar(barra, alvo)
    local holding = false
    local starting, previous

    barra.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            holding = true
            starting = input.Position
            previous = alvo.Position
        end
    end)

    barra.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            holding = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not holding or not starting then
            return
        end

        if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
            return
        end

        local offset = input.Position - starting
        alvo.Position = UDim2.new(previous.X.Scale, previous.X.Offset + offset.X, previous.Y.Scale, previous.Y.Offset + offset.Y)
    end)
end

for _, s in ipairs(sliders) do
    local active = false

    local function update(input)
        local bar = item[s.bar]
        local fraction = math.clamp((input.Position.X - bar.AbsolutePosition.X) / math.max(1, bar.AbsoluteSize.X), 0, 1)
        local value = math.floor(s.min + fraction * (s.max - s.min))
        item[s.label].Text = tostring(value)
        item[s.fill].Size = UDim2.new(fraction, 0, 1, 0)
        item[s.handle].Position = UDim2.new(fraction, -item[s.handle].Size.X.Offset / 2, 0.5, -item[s.handle].Size.Y.Offset / 2)
        emit("slider", s.name, value)
    end

    item[s.handle].InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            active = true
            update(input)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if active and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            active = false
        end
    end)
end
